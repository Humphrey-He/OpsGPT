### 3.2 核心代码 - 文档解析

**`pkg/parser/chunker.go`:**

```go
package parser

import (
    "strings"
)

type Chunk struct {
    Content   string
    Metadata  map[string]string
    TokenCount int
}

type Chunker struct {
    chunkSize    int
    overlapSize  int
}

func NewChunker(chunkSize, overlapSize int) *Chunker {
    return &Chunker{
        chunkSize:   chunkSize,
        overlapSize: overlapSize,
    }
}

func (c *Chunker) Chunk(text string, metadata map[string]string) []Chunk {
    // 简单按段落分块
    paragraphs := strings.Split(text, "\n\n")
    var chunks []Chunk
    var current strings.Builder
    
    for _, p := range paragraphs {
        p = strings.TrimSpace(p)
        if p == "" {
            continue
        }
        
        if current.Len()+len(p) > c.chunkSize {
            if current.Len() > 0 {
                chunks = append(chunks, Chunk{
                    Content:   current.String(),
                    Metadata:  metadata,
                    TokenCount: estimateTokens(current.String()),
                })
                // 保留 overlap
                overlap := current.String()
                if len(overlap) > c.overlapSize {
                    overlap = overlap[len(overlap)-c.overlapSize:]
                }
                current.Reset()
                current.WriteString(overlap)
            }
        }
        current.WriteString(p)
        current.WriteString("\n\n")
    }
    
    if current.Len() > 0 {
        chunks = append(chunks, Chunk{
            Content:   current.String(),
            Metadata:  metadata,
            TokenCount: estimateTokens(current.String()),
        })
    }
    
    return chunks
}

func estimateTokens(text string) int {
    // 粗略估算: 中文 2 字符 ≈ 1 token, 英文 4 字符 ≈ 1 token
    chinese := 0
    for _, r := range text {
        if r > 127 {
            chinese++
        }
    }
    english := len(text) - chinese
    return chinese/2 + english/4
}
```

### 3.3 核心代码 - 向量存储

**`pkg/vector/qdrant.go`:**

```go
package vector

import (
    "context"
    "fmt"
    
    "github.com/qdrant/go-client/v2"
    "github.com/qdrant/go-client/v2/qdrant"
)

type QdrantStore struct {
    client     *qdrant.Client
    collection string
    dimension  int
}

func NewQdrantStore(url, collection string, dimension int) (*QdrantStore, error) {
    client, err := qdrant.NewClient(&qdrant.Config{
        Host: url,
    })
    if err != nil {
        return nil, err
    }
    
    return &QdrantStore{
        client:     client,
        collection: collection,
        dimension:  dimension,
    }, nil
}

func (s *QdrantStore) CreateCollection(ctx context.Context) error {
    _, err := s.client.CreateCollection(ctx, &qdrant.CreateCollection{
        CollectionName: s.collection,
        VectorsConfig: qdrant.NewSingleVectorParams(&qdrant.VectorParams{
            Size:     uint64(s.dimension),
            Distance: qdrant.Distance_Cosine,
        }),
    })
    return err
}

func (s *QdrantStore) UpsertPoints(ctx context.Context, points []Point) error {
    payloads := make([]*qdrant.Payload, len(points))
    vectors := make([][]float32, len(points))
    
    for i, p := range points {
        payloads[i] = qdrant.NewPayload(p.Payload)
        vectors[i] = p.Vector
    }
    
    _, err := s.client.Upsert(ctx, &qdrant.UpsertPoints{
        CollectionName: s.collection,
        Points: []*qdrant.PointStruct{
            {
                Id:       nil,
                Vectors:  vectors,
                Payload:  payloads,
            },
        },
    })
    return err
}

type Point struct {
    ID       string
    Vector   []float32
    Payload  map[string]interface{}
}
```

### 3.4 核心代码 - LLM 客户端

**`pkg/llm/ollama.go`:**

```go
package llm

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "time"
)

type OllamaClient struct {
    baseURL string
    model   string
    client  *http.Client
}

func NewOllamaClient(baseURL, model string) *OllamaClient {
    return &OllamaClient{
        baseURL: baseURL,
        model:   model,
        client: &http.Client{
            Timeout: 120 * time.Second,
        },
    }
}

type ChatRequest struct {
    Model    string        `json:"model"`
    Messages []Message     `json:"messages"`
    Stream   bool          `json:"stream"`
}

type Message struct {
    Role    string `json:"role"`
    Content string `json:"content"`
}

func (c *OllamaClient) Chat(ctx context.Context, messages []Message, stream bool) (*ChatResponse, error) {
    reqBody := ChatRequest{
        Model:    c.model,
        Messages: messages,
        Stream:   stream,
    }
    
    body, _ := json.Marshal(reqBody)
    
    req, err := http.NewRequestWithContext(ctx, "POST", 
        c.baseURL+"/api/chat", bytes.NewReader(body))
    if err != nil {
        return nil, err
    }
    req.Header.Set("Content-Type", "application/json")
    
    resp, err := c.client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    respBody, err := io.ReadAll(resp.Body)
    if err != nil {
        return nil, err
    }
    
    var result ChatResponse
    if err := json.Unmarshal(respBody, &result); err != nil {
        return nil, err
    }
    
    return &result, nil
}

type ChatResponse struct {
    Model              string   `json:"model"`
    Message            Message  `json:"message"`
    Done               bool    `json:"done"`
}

func (c *OllamaClient) GenerateEmbeddings(ctx context.Context, text string) ([]float32, error) {
    reqBody := map[string]string{
        "model": c.model,
        "prompt": text,
    }
    
    body, _ := json.Marshal(reqBody)
    
    req, err := http.NewRequestWithContext(ctx, "POST",
        c.baseURL+"/api/embeddings", bytes.NewReader(body))
    if err != nil {
        return nil, err
    }
    req.Header.Set("Content-Type", "application/json")
    
    resp, err := c.client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    var result map[string]interface{}
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, err
    }
    
    embedding, ok := result["embedding"].([]interface{})
    if !ok {
        return nil, fmt.Errorf("no embedding in response")
    }
    
    vec := make([]float32, len(embedding))
    for i, v := range embedding {
        vec[i], _ = v.(float64)
    }
    
    return vec, nil
}
```

### 3.5 核心代码 - RAG 链路

**`internal/rag/chain.go`:**

```go
package rag

import (
    "context"
    "fmt"
    
    "GopherSentinel/pkg/llm"
    "GopherSentinel/pkg/parser"
    "GopherSentinel/pkg/vector"
)

type RAGChain struct {
    llm      *llm.OllamaClient
    vector   *vector.QdrantStore
    chunker  *parser.Chunker
}

func NewRAGChain(llm *llm.OllamaClient, vector *vector.QdrantStore) *RAGChain {
    return &RAGChain{
        llm:     llm,
        vector:  vector,
        chunker: parser.NewChunker(500, 50),
    }
}

func (r *RAGChain) IndexDocument(ctx context.Context, doc *parser.Document) error {
    chunks := r.chunker.Chunk(doc.Content, doc.Metadata)
    
    for _, chunk := range chunks {
        embedding, err := r.llm.GenerateEmbeddings(ctx, chunk.Content)
        if err != nil {
            return fmt.Errorf("embedding failed: %w", err)
        }
        
        point := vector.Point{
            ID:      fmt.Sprintf("%s-%d", doc.ID, len(chunks)),
            Vector:  embedding,
            Payload: map[string]interface{}{
                "content":  chunk.Content,
                "metadata": chunk.Metadata,
            },
        }
        
        if err := r.vector.UpsertPoints(ctx, []vector.Point{point}); err != nil {
            return fmt.Errorf("upsert failed: %w", err)
        }
    }
    
    return nil
}

func (r *RAGChain) Query(ctx context.Context, question string, history []llm.Message) (string, error) {
    // 1. 生成问题向量
    queryEmbedding, err := r.llm.GenerateEmbeddings(ctx, question)
    if err != nil {
        return "", err
    }
    
    // 2. 检索相关文档 (简化版，后续补充)
    // 3. 构建 prompt
    systemPrompt := `你是一个智能助手。请根据提供的上下文信息回答用户的问题。
如果上下文中没有相关信息，请如实告知。`
    
    messages := []llm.Message{
        {Role: "system", Content: systemPrompt},
    }
    
    // 添加历史
    messages = append(messages, history...)
    
    // 添加上下文和当前问题
    messages = append(messages, llm.Message{
        Role:    "user",
        Content: fmt.Sprintf("上下文信息：\n%s\n\n问题：%s", "[检索到的相关文档]", question),
    })
    
    // 4. 调用 LLM
    resp, err := r.llm.Chat(ctx, messages, false)
    if err != nil {
        return "", err
    }
    
    return resp.Message.Content, nil
}
```

```bash

ollama serve

# 设置环境变量
```

```bash
# 检查服务
docker logs <container_id>

# 手动创建 Collection
curl -X PUT http://localhost:6333/collections/docs \
  -H 'Content-Type: application/json' \
  --data '{"vectors": {"size": 768, "distance": "Cosine"}}'
```

